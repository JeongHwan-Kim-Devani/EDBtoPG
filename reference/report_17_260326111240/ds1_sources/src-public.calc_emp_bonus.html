    -- 변수 선언부 (Oracle 데이터 타입인 VARCHAR2 사용 가능)
    v_emp_name    VARCHAR2(50);
    v_base_salary NUMERIC;
BEGIN
    -- 1. 직원 정보 조회 (암시적 커서)
    SELECT emp_name, base_salary 
    INTO v_emp_name, v_base_salary
    FROM epas_emp_contracts
    WHERE contract_id = p_contract_id;
    -- 2. 새로운 급여 계산
    p_new_salary := v_base_salary + (v_base_salary * p_bonus_rate / 100);
    -- 3. 결과 로그 출력 (Oracle 호환 내장 패키지 사용)
    DBMS_OUTPUT.PUT_LINE('====================================');
    DBMS_OUTPUT.PUT_LINE('직원명: ' || v_emp_name);
    DBMS_OUTPUT.PUT_LINE('기존 급여: ' || v_base_salary);
    DBMS_OUTPUT.PUT_LINE('보너스 적용 급여: ' || p_new_salary);
    DBMS_OUTPUT.PUT_LINE('====================================');
EXCEPTION
    -- 예외 처리부 (Oracle 호환 예외 코드)
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('오류: 사번 ' || p_contract_id || '에 해당하는 직원이 없습니다.');
        p_new_salary := 0;
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('알 수 없는 오류 발생: ' || SQLERRM);
        p_new_salary := -1;
END
