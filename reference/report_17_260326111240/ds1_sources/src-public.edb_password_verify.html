BEGIN
    -- 1. 최소 길이 체크 (8자 미만)
    IF length(p_password) < 8 THEN
        RAISE EXCEPTION 'Password is too short (minimum 8 characters)';
    END IF;
    -- 2. 사용자 ID와 동일한지 체크
    IF p_password = p_username THEN
        RAISE EXCEPTION 'Password cannot be the same as the username';
    END IF;
    
    -- 3. (선택사항) 이전 비밀번호와 동일한지 체크는 EDB가 PASSWORD_REUSE 옵션으로 알아서 하므로 여기서 굳이 안 해도 됩니다.
    RETURN true; -- 통과 시 true 반환
END;
