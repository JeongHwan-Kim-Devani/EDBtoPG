 SELECT contract_id,
    emp_name,
    base_salary,
    commission AS original_commission,
    nvl(commission, 0) AS actual_commission,
    (base_salary + nvl(commission, 0)) AS total_salary,
    contract_date,
    add_months(contract_date, (12)::numeric) AS contract_expiration_date,
    systimestamp AS current_check_time
   FROM epas_emp_contracts;
