    v_enc_raw    RAW(32767);
    v_dec_raw    RAW(32767);
    v_key_raw    RAW(32);
    
    -- [수정] 암호화와 동일하게 4358 사용
    v_algo_type  PLS_INTEGER := 4358;
BEGIN
    IF p_enc_hex IS NULL THEN
        RETURN NULL;
    END IF;
    v_enc_raw := HEXTORAW(p_enc_hex);
    -- [수정] 복호화 키 생성도 동일하게 MD5(2) 사용
    v_key_raw := SYS.DBMS_CRYPTO.HASH(SYS.UTL_RAW.CAST_TO_RAW(p_key), 2);
    v_dec_raw := SYS.DBMS_CRYPTO.DECRYPT(
        src => v_enc_raw,
        typ => v_algo_type,
        key => v_key_raw
    );
    RETURN SYS.UTL_RAW.CAST_TO_VARCHAR2(v_dec_raw);
END
