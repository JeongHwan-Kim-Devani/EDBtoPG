DECLARE
    v_plain_text    VARCHAR2(200);
    v_plain_raw     RAW(2000);
    v_enc_raw       RAW(2000);
    v_hash_raw      RAW(64);
    v_key_raw       RAW(32);
BEGIN
    FOR i IN 1..10 LOOP
        v_key_raw := DBMS_CRYPTO.RANDOMBYTES(8);
        
        v_hash_raw := DBMS_CRYPTO.HASH(
            UTL_RAW.CAST_TO_RAW('my_secret_pwd_' || i),
            2
        );
        
        v_plain_text := '이것은 user_' || i || '의 1급 비밀 메시지입니다.';
        v_plain_raw  := UTL_RAW.CAST_TO_RAW(v_plain_text);
        
        v_enc_raw := DBMS_CRYPTO.ENCRYPT(
            v_plain_raw,
            4353,
            v_key_raw
        );
        
        INSERT INTO sc_crypto.user_secure_data (
            username, pwd_hash, encryption_key, secure_message
        ) VALUES (
            'user_' || i, v_hash_raw, v_key_raw, v_enc_raw
        );
    END LOOP;
    COMMIT;
END
