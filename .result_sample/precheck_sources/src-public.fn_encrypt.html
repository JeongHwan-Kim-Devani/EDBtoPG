    -- [수정] 4358 = AES128(6) + CBC(256) + PKCS5(4096)
    v_algo_type  PLS_INTEGER := 4358; 
BEGIN
    IF p_plain_text IS NULL THEN
        RETURN NULL;
    END IF;
    v_src_raw := SYS.UTL_RAW.CAST_TO_RAW(p_plain_text);
    -- [수정] 해시 알고리즘을 '2' (MD5)로 변경
    -- 이유: MD5는 16바이트를 반환하므로 AES-128(16바이트 키)과 완벽하게 매칭됨
    v_key_raw := SYS.DBMS_CRYPTO.HASH(SYS.UTL_RAW.CAST_TO_RAW(p_key), 2);
    v_enc_raw := SYS.DBMS_CRYPTO.ENCRYPT(
        src => v_src_raw,
        typ => v_algo_type,
        key => v_key_raw
    );
    RETURN RAWTOHEX(v_enc_raw);
END
