DECLARE
    v_date DATE := sysdate;
    v_count INT;
BEGIN
    -- 오라클 전용 뷰 (user_tables, dba_objects, v$session) 및 함수 (nvl, decode) 사용
    SELECT nvl(count(*), 0) INTO v_count FROM user_tables;
    IF decode(v_count, 0, 'ZERO', 'MANY') = 'ZERO' THEN
        SELECT count(*) INTO v_count FROM dba_objects;
    END IF;
    PERFORM * FROM v$session;
END;
